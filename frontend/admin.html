(() => { "use strict"; const API_BASE = "https://smart-parking-finder-system-tlzt.onrender.com/api"; let adminToken = localStorage.getItem("adminToken") || null; let parkingAreas = []; // DOM Elements const adminLoginView = document.getElementById("adminLoginView"); const adminControlView = document.getElementById("adminControlView"); const adminLoginForm = document.getElementById("adminLoginForm"); const adminUsernameInput = document.getElementById("adminUsername"); const adminPasswordInput = document.getElementById("adminPassword"); const adminAuthFeedback = document.getElementById("adminAuthFeedback"); const adminLogoutBtn = document.getElementById("adminLogoutBtn"); const adminOverviewBody = document.getElementById("adminOverviewBody"); const adminForm = document.getElementById("adminForm"); const adminLotSelect = document.getElementById("adminLotSelect"); const adminSpacesInput = document.getElementById("adminSpacesInput"); const adminFeedback = document.getElementById("adminFeedback"); // API Helper async function apiFetch(path, options = {}) { const headers = { "Content-Type": "application/json", ...(options.headers || {}) }; if (adminToken) { headers["Authorization"] = `Bearer ${adminToken}`; } let res; try { res = await fetch(`${API_BASE}${path}`, { ...options, headers }); } catch (networkErr) { throw new Error("Cannot connect to server. Please check your backend."); } const data = await res.json().catch(() => ({})); if (!res.ok) throw new Error(data.error || "Request failed."); return data; } // View Switches function showLoggedInView() { if (adminLoginView) adminLoginView.hidden = true; if (adminControlView) adminControlView.hidden = false; loadDashboard(); } function showLoggedOutView() { if (adminLoginView) adminLoginView.hidden = false; if (adminControlView) adminControlView.hidden = true; if (adminLoginForm) adminLoginForm.reset(); if (adminAuthFeedback) adminAuthFeedback.textContent = ""; } // Login Handler async function handleLogin(e) { e.preventDefault(); const identifier = adminUsernameInput.value.trim(); const password = adminPasswordInput.value; try { const data = await apiFetch("/auth/admin/login", { method: "POST", body: JSON.stringify({ username: identifier, email: identifier, password }), }); adminToken = data.token; localStorage.setItem("adminToken", adminToken); if (adminAuthFeedback) adminAuthFeedback.textContent = ""; showLoggedInView(); } catch (err) { if (adminAuthFeedback) { adminAuthFeedback.dataset.tone = "error"; adminAuthFeedback.textContent = err.message; } } } // Logout Handler function handleLogout() { adminToken = null; localStorage.removeItem("adminToken"); showLoggedOutView(); } // Load Parking Dashboard async function loadDashboard() { try { parkingAreas = await apiFetch("/parking-areas"); renderOverviewTable(); populateLotSelect(); } catch (err) { if (adminFeedback) { adminFeedback.dataset.tone = "error"; adminFeedback.textContent = "Failed to load parking areas."; } } } // Render Table & Options function renderOverviewTable() { if (!adminOverviewBody) return; adminOverviewBody.innerHTML = parkingAreas .map( (area) => ` ${area.name} ${area.availableSpaces} ${area.totalSpaces} ` ) .join(""); } function populateLotSelect() { if (!adminLotSelect) return; adminLotSelect.innerHTML = parkingAreas .map((area) => ``) .join(""); if (parkingAreas.length > 0) { updateSpacesInputValue(); } } function updateSpacesInputValue() { const selectedId = adminLotSelect.value; const selectedArea = parkingAreas.find((a) => a.id === selectedId); if (selectedArea && adminSpacesInput) { adminSpacesInput.value = selectedArea.availableSpaces; adminSpacesInput.max = selectedArea.totalSpaces; } } // Update Space Count Handler async function handleUpdateAvailability(e) { e.preventDefault(); const id = adminLotSelect.value; const availableSpaces = parseInt(adminSpacesInput.value, 10); try { await apiFetch(`/parking-areas/${id}`, { method: "PUT", body: JSON.stringify({ availableSpaces }), }); if (adminFeedback) { adminFeedback.dataset.tone = "success"; adminFeedback.textContent = "Availability updated successfully!"; } await loadDashboard(); adminLotSelect.value = id; updateSpacesInputValue(); setTimeout(() => { if (adminFeedback) adminFeedback.textContent = ""; }, 3000); } catch (err) { if (adminFeedback) { adminFeedback.dataset.tone = "error"; adminFeedback.textContent = err.message; } } } // Event Listeners Initialization function init() { if (adminLoginForm) adminLoginForm.addEventListener("submit", handleLogin); if (adminLogoutBtn) adminLogoutBtn.addEventListener("click", handleLogout); if (adminForm) adminForm.addEventListener("submit", handleUpdateAvailability); if (adminLotSelect) adminLotSelect.addEventListener("change", updateSpacesInputValue); if (adminToken) { showLoggedInView(); } else { showLoggedOutView(); } } document.addEventListener("DOMContentLoaded", init); })();
